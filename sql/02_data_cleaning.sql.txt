-- ------------------------------------------------------------
-- DATA CLEANING AND PREPROCESSING
-- ------------------------------------------------------------

-- Step 1: Handling foreign characters (example: specific show_id)
SELECT * 
FROM netflix_raw 
WHERE show_id = 's5023';

-- Step 2: Identifying and removing duplicates based on title
SELECT show_id, COUNT(*) AS duplicate_count
FROM netflix_raw
GROUP BY title
HAVING COUNT(*) > 1;

-- Adjusting SQL mode to allow for more flexible GROUP BY behavior
SET sql_mode = (SELECT REPLACE(@@sql_mode, 'ONLY_FULL_GROUP_BY', ''));

-- Step 3: Handle duplicates by concatenating title and type fields
SELECT * 
FROM netflix_raw 
WHERE CONCAT(UPPER(title), type) IN (
    SELECT CONCAT(UPPER(title), type)
    FROM netflix_raw
    GROUP BY UPPER(title), type
    HAVING COUNT(*) > 1
)
ORDER BY title;

-- Step 4: Remove duplicates by selecting the first instance using ROW_NUMBER
WITH cte AS (
    SELECT *,
           ROW_NUMBER() OVER (PARTITION BY title, type ORDER BY show_id) AS rn
    FROM netflix_raw
)
SELECT *
FROM cte
WHERE rn = 1;

-- ------------------------------------------------------------
-- CREATING AUXILIARY TABLES (Directors, Countries, Casts, Genres)
-- ------------------------------------------------------------

-- Step 5: Create table for directors (splitting multiple directors)
DROP TABLE IF EXISTS netflix_directors;
CREATE TABLE netflix_directors AS
WITH RECURSIVE numbers AS (
    SELECT 1 AS n
    UNION ALL
    SELECT n + 1 FROM numbers WHERE n < 100
)
SELECT 
    show_id,
    TRIM(SUBSTRING_INDEX(SUBSTRING_INDEX(director, ',', n), ',', -1)) AS director
FROM 
    netflix_raw
JOIN 
    numbers ON n <= 1 + LENGTH(director) - LENGTH(REPLACE(director, ',', ''))
WHERE director IS NOT NULL;

-- Step 6: Create table for countries (splitting multiple countries)
DROP TABLE IF EXISTS netflix_countries;
CREATE TABLE netflix_countries AS
WITH RECURSIVE numbers AS (
    SELECT 1 AS n
    UNION ALL
    SELECT n + 1 FROM numbers WHERE n < 100
)
SELECT 
    show_id,
    TRIM(SUBSTRING_INDEX(SUBSTRING_INDEX(country, ',', n), ',', -1)) AS country
FROM 
    netflix_raw
JOIN 
    numbers ON n <= 1 + LENGTH(country) - LENGTH(REPLACE(country, ',', ''))
WHERE country IS NOT NULL;

-- Step 7: Create table for casts (splitting multiple casts)
DROP TABLE IF EXISTS netflix_casts;
CREATE TABLE netflix_casts AS
WITH RECURSIVE numbers AS (
    SELECT 1 AS n
    UNION ALL
    SELECT n + 1 FROM numbers WHERE n < 100
)
SELECT 
    show_id,
    TRIM(SUBSTRING_INDEX(SUBSTRING_INDEX(cast, ',', n), ',', -1)) AS cast
FROM 
    netflix_raw
JOIN 
    numbers ON n <= 1 + LENGTH(cast) - LENGTH(REPLACE(cast, ',', ''))
WHERE cast IS NOT NULL;

-- Step 8: Create table for listed_in (splitting multiple genres)
DROP TABLE IF EXISTS netflix_listedins;
CREATE TABLE netflix_listedins AS
WITH RECURSIVE numbers AS (
    SELECT 1 AS n
    UNION ALL
    SELECT n + 1 FROM numbers WHERE n < 100
)
SELECT 
    show_id,
    TRIM(SUBSTRING_INDEX(SUBSTRING_INDEX(listed_in, ',', n), ',', -1)) AS genre
FROM 
    netflix_raw
JOIN 
    numbers ON n <= 1 + LENGTH(listed_in) - LENGTH(REPLACE(listed_in, ',', ''))
WHERE listed_in IS NOT NULL;

-- ------------------------------------------------------------
-- FINAL DATA CLEANING AND CREATING THE FINAL 'netflix' TABLE
-- ------------------------------------------------------------

-- Step 9: Create 'netflix' table with cleaned data (remove duplicates, handle NULLs)
CREATE TABLE netflix AS
WITH cte AS (
    SELECT *,
           ROW_NUMBER() OVER (PARTITION BY title, type ORDER BY show_id) AS rn
    FROM netflix_raw
)
SELECT 
    show_id, 
    type, 
    title,
    STR_TO_DATE(date_added, '%M %d, %Y') AS date_added,  -- Convert date format
    release_year,
    rating, 
    CASE 
        WHEN duration IS NULL THEN rating  -- Use rating if duration is NULL
        ELSE duration 
    END AS duration, 
    description
FROM cte
WHERE rn = 1;

-- Step 10: Query to check for missing countries
SELECT show_id, country
FROM netflix_raw
WHERE country IS NULL;

-- Step 11: Insert missing country data by matching with directors
INSERT INTO netflix_countries
SELECT nr.show_id, m.country
FROM netflix_raw nr
INNER JOIN (
    SELECT nd.director, nr2.country
    FROM netflix_raw nr2
    INNER JOIN netflix_directors nd ON nr2.show_id = nd.show_id
    WHERE nr2.country IS NOT NULL
    GROUP BY nd.director, nr2.country
) AS m ON nr.director = m.director
WHERE nr.country IS NULL;

-- Step 12: Query for specific director (example: 'Ahishor Solomon')
SELECT * 
FROM netflix_raw

-- Step 13: Get director and associated country details (grouped and ordered)
SELECT director, country
FROM netflix_countries nc
INNER JOIN netflix_directors nd ON nc.show_id = nd.show_id
GROUP BY director, country
ORDER BY director;
